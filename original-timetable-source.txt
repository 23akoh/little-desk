    <script>
        const ALL_DAYS = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"];
        const EXAM_BOARDS = ["AQA", "Edexcel", "OCR", "WJEC", "Eduqas", "CCEA", "CIE"];
        let currentDailyViewIndex = 0;
        
        let appUserProfile = null;
        let homeworkEntriesList = [];
        let globalBase64PFP = "";

        function parseTimeToMinutes(tStr) {
            if(!tStr) return 0;
            const parts = tStr.split(':');
            return (parseInt(parts[0]) * 60) + parseInt(parts[1]);
        }

        function formatMinutesToTime(totalMins) {
            const hrs = Math.floor(totalMins / 60) % 24;
            const mins = totalMins % 60;
            return `${String(hrs).padStart(2, '0')}:${String(mins).padStart(2, '0')}`;
        }

        function processPFP(event) {
            const file = event.target.files[0];
            if(file) {
                const reader = new FileReader();
                reader.onload = e => {
                    globalBase64PFP = e.target.result;
                    document.getElementById('setup-pfp-preview').src = globalBase64PFP;
                    document.getElementById('dynamic-favicon').setAttribute('href', globalBase64PFP);
                };
                reader.readAsDataURL(file);
            }
        }

        function appendLessonRowToSetup(day, subject) {
            const id = 'lesson-row-' + Date.now() + Math.floor(Math.random() * 100);
            const div = document.createElement('div');
            div.id = id;
            div.className = "flex flex-wrap items-center gap-1 bg-white p-1.5 border rounded-lg shadow-xs";
            
            let daysHtml = ALL_DAYS.slice(0, 5).map(d => `<option value="${d}" ${day===d?'selected':''}>${d}</option>`).join('');
            div.innerHTML = `
                <select class="text-xs p-1.5 border rounded w-1/3">${daysHtml}</select>
                <input type="text" list="subject-suggestions" placeholder="Subject Name" value="${subject}" class="text-xs p-1.5 border rounded flex-1" required>

        function generateSmartTimetable(profile) {
            const weekDataMatrix = {};

            ALL_DAYS.forEach((day, dIdx) => {
                const dayPlan = [];
                const isWeekend = (day === "Saturday" || day === "Sunday");
                const timelineMap = [];

                const startWakeMins = parseTimeToMinutes(profile.wakeTime);
                const morningPrepEnd = startWakeMins + profile.morningReady;
                
                dayPlan.push({ type: "Routine", title: "☀️ Wake & Prep", start: startWakeMins, end: morningPrepEnd, display: `${profile.wakeTime} - ${formatMinutesToTime(morningPrepEnd)}`, bg: "bg-amber-50 border-amber-200", text: "text-amber-900" });
                timelineMap.push({ start: startWakeMins, end: morningPrepEnd });

                let schoolStartMins = parseTimeToMinutes(profile.schoolStart);
                let schoolEndMins = parseTimeToMinutes(profile.schoolEnd);
                
                if(!isWeekend) {
                    // Block school travel time buffer BEFORE school
                    let preSchoolTravelStart = Math.max(0, schoolStartMins - 30);
                    timelineMap.push({ start: preSchoolTravelStart, end: schoolStartMins });
                    dayPlan.push({ type: "Buffer", title: "🚗 Travel / Rest Buffer", start: preSchoolTravelStart, end: schoolStartMins, display: `${formatMinutesToTime(preSchoolTravelStart)} - ${profile.schoolStart}`, bg: "bg-gray-100 border-gray-200 text-gray-500 italic", text: "text-gray-500" });

                    // Block School core hours
                    dayPlan.push({ type: "School", title: "🏫 Core School Hours", start: schoolStartMins, end: schoolEndMins, display: `${profile.schoolStart} - ${profile.schoolEnd}`, bg: "bg-blue-50 border-blue-200", text: "text-blue-900" });
                    timelineMap.push({ start: schoolStartMins, end: schoolEndMins });

                    // Block school travel time buffer AFTER school
                    let postSchoolTravelEnd = schoolEndMins + 30;
                    timelineMap.push({ start: schoolEndMins, end: postSchoolTravelEnd });
                    dayPlan.push({ type: "Buffer", title: "🚗 Travel / Rest Buffer", start: schoolEndMins, end: postSchoolTravelEnd, display: `${profile.schoolEnd} - ${formatMinutesToTime(postSchoolTravelEnd)}`, bg: "bg-gray-100 border-gray-200 text-gray-500 italic", text: "text-gray-500" });
                }

                profile.clubs.filter(c => c.day === day).forEach(club => {
                    const cStart = parseTimeToMinutes(club.start);
                    const cEnd = parseTimeToMinutes(club.end);
                    
                    // Buffer BEFORE club
                    let preClubStart = Math.max(0, cStart - 30);
                    timelineMap.push({ start: preClubStart, end: cStart });
                    dayPlan.push({ type: "Buffer", title: `🚗 Buffer: ${club.name}`, start: preClubStart, end: cStart, display: `${formatMinutesToTime(preClubStart)} - ${club.start}`, bg: "bg-gray-100 border-gray-200 text-gray-400 italic", text: "text-gray-500" });

                    // Club event itself
                    dayPlan.push({ type: "Club", title: `🏆 Club: ${club.name}`, start: cStart, end: cEnd, display: `${club.start} - ${club.end}`, bg: "bg-rose-50 border-rose-200", text: "text-rose-900" });
                    timelineMap.push({ start: cStart, end: cEnd });

                    // Buffer AFTER club
                    let postClubEnd = cEnd + 30;
                    timelineMap.push({ start: cEnd, end: postClubEnd });
                    dayPlan.push({ type: "Buffer", title: `🚗 Buffer: ${club.name}`, start: cEnd, end: postClubEnd, display: `${club.end} - ${formatMinutesToTime(postClubEnd)}`, bg: "bg-gray-100 border-gray-200 text-gray-400 italic", text: "text-gray-500" });
                });

                const bedMins = parseTimeToMinutes(profile.bedtime);
                const restStartMins = bedMins - profile.eveningRest;
                dayPlan.push({ type: "Routine", title: "🛋️ Evening Rest", start: restStartMins, end: bedMins, display: `${formatMinutesToTime(restStartMins)} - ${profile.bedtime}`, bg: "bg-emerald-50 border-emerald-200", text: "text-emerald-900" });
                timelineMap.push({ start: restStartMins, end: bedMins });

                dayPlan.push({ type: "Sleep", title: "🌙 Sleep Window", start: bedMins, end: 1440, display: `${profile.bedtime} onwards`, bg: "bg-slate-100 border-slate-200", text: "text-slate-600" });
                timelineMap.push({ start: bedMins, end: 1440 });
                timelineMap.push({ start: 0, end: startWakeMins });

                let smartDailyPool = [];
                let lessonSubjectsToday = profile.lessons ? profile.lessons.filter(l => l.day === day).map(l => l.subject) : [];
                
                homeworkEntriesList.forEach(hw => {
                    if(!hw.completed) {
                        let hwDate = new Date(hw.date);
                        let hwDayName = hwDate.getDay() === 0 ? "Sunday" : ALL_DAYS[hwDate.getDay() - 1];
                        let targetDayIndex = ALL_DAYS.indexOf(hwDayName) - 1; 
                        if(targetDayIndex < 0) targetDayIndex = 6;
                        
                        if(dIdx === targetDayIndex) {
                            let duration = parseInt(hw.duration) || 30;
                            let scheduled = attemptFitInterval(timelineMap, morningPrepEnd, restStartMins, duration);
                            if(scheduled) {
                                dayPlan.push({ type: "Homework", title: `📝 HW: ${hw.title}`, reason: "Reason: Imminent Deadline", start: scheduled.start, end: scheduled.end, display: `${formatMinutesToTime(scheduled.start)} - ${formatMinutesToTime(scheduled.end)}`, bg: "bg-indigo-50 border-indigo-200 font-bold", text: "text-indigo-900", checkable: true, hwRef: hw.title });
                                timelineMap.push({ start: scheduled.start, end: scheduled.end });
                            }
                        }
                    }
                });

                profile.subjects.forEach(s => {
                    const weighting = s.priority ? 3 : 1;
                    for(let i=0; i<weighting; i++) {
                        if(lessonSubjectsToday.includes(s.name)) {
                            smartDailyPool.unshift({ name: s.name, reason: `Reason: Studied in today's classes (Spaced Repetition)` });
                        } else {
                            smartDailyPool.push({ name: s.name, reason: "Reason: General priority rotation" });
                        }
                    }
                });
                if(smartDailyPool.length === 0) smartDailyPool.push({ name: "General Review", reason: "Maintain study habits" });

                let quota = isWeekend ? profile.wndCount : profile.wkCount;
                for(let q=0; q<quota; q++) {
                    let scheduledSlot = attemptFitInterval(timelineMap, morningPrepEnd, restStartMins, profile.slotLen);
                    if(scheduledSlot) {
                        let currentSub = smartDailyPool[q % smartDailyPool.length];
                        dayPlan.push({ type: "Revision", title: `📚 Revise: ${currentSub.name}`, reason: currentSub.reason, start: scheduledSlot.start, end: scheduledSlot.end, display: `${formatMinutesToTime(scheduledSlot.start)} - ${formatMinutesToTime(scheduledSlot.end)}`, bg: "bg-white border shadow-xs border-l-4", text: "text-gray-800", checkable: true });
                        timelineMap.push({ start: scheduledSlot.start, end: scheduledSlot.end + profile.breakLen });
                    }
                }

                dayPlan.sort((a,b) => a.start - b.start);
                weekDataMatrix[day] = dayPlan;
            });

            const gridBody = document.getElementById('grid-body');
            gridBody.innerHTML = '';
            let maxRowLimit = Math.max(...Object.values(weekDataMatrix).map(arr => arr.length));
            
            for(let r=0; r<maxRowLimit; r++) {
                let tr = document.createElement('tr');
                ALL_DAYS.forEach(day => {
                    let td = document.createElement('td');
                    td.className = "p-1.5 border-r border-b align-top w-[14.28%] overflow-hidden";
                    let item = weekDataMatrix[day][r];
                    if(item) {
                        let chkBox = item.checkable ? `<input type="checkbox" class="mr-1.5 mt-0.5 w-3.5 h-3.5 flex-shrink-0 dynamic-accent rounded" onchange="handleTaskTick('${item.type}', '${item.hwRef || ''}', this.checked)">` : '';
                        let reasonHtml = item.reason ? `<span class="block text-[8px] text-purple-600 mt-0.5 italic leading-tight">${item.reason}</span>` : '';
                        
                        td.innerHTML = `
                            <div class="${item.bg} ${item.text} p-2 rounded text-[11px] flex items-start h-full border overflow-hidden max-w-[150px]">
                                ${chkBox}
                                <div class="overflow-hidden truncate whitespace-normal">
                                    <span class="block font-bold leading-tight break-words">${item.title}</span>
                                    ${reasonHtml}
                                    <span class="block text-[9px] opacity-75 mt-1 font-mono">${item.display}</span>
                                </div>
                            </div>`;
                    }
                    tr.appendChild(td);
                });
                gridBody.appendChild(tr);
            }

            window.currentWeekDataMatrix = weekDataMatrix;
        }

        function attemptFitInterval(timeline, searchStart, searchEnd, duration) {
            for(let timeCursor = searchStart; timeCursor <= (searchEnd - duration); timeCursor += 5) {
                let dynamicOverlap = false;
                for(let b=0; b<timeline.length; b++) {
                    if(timeCursor < timeline[b].end && (timeCursor + duration) > timeline[b].start) {
                        dynamicOverlap = true;
                        break;
                    }
                }
                if(!dynamicOverlap) {
                    return { start: timeCursor, end: timeCursor + duration };
                }
            }
            return null;
        }

        }
